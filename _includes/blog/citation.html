Citation

Cited as:

{{ site.blog.cite_author }}. ({{ month }} {{ year }}). {{ page.title }}. {{ site.blog.title }}. {{ page.url | absolute_url }}.

Or

@article{{ "{" }}{{ site.blog.cite_key }}{{ year }}{{ key }},
  title   = "{{ page.title }}",
  author  = "{{ site.blog.cite_author }}",
  journal = "{{ site.url | remove: 'https://' | remove: 'http://' }}",
  year    = "{{ year }}",
  month   = "{{ month }}",
  url     = "{{ page.url | absolute_url }}"
}
